import argparse,subprocess
from pathlib import Path
p=argparse.ArgumentParser();p.add_argument('--repo',type=Path,required=True);p.add_argument('--cli',required=True);a=p.parse_args()
for example in sorted((a.repo/'examples').iterdir()):
 cmd=[a.cli,'inspect','--input',str(example/'input.json'),'--proposals',str(example/'proposals.json'),'--format','markdown','--output',str(example/'expected_report.md')]
 result=subprocess.run(cmd,check=True,capture_output=True,text=True)
 print(example.name,'exit',result.returncode)
